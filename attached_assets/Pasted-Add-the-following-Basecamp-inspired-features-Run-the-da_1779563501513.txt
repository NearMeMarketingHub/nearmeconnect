=== DATABASE MIGRATION (run first) ===

CREATE TABLE IF NOT EXISTS public.notepads (
    id character varying DEFAULT gen_random_uuid() NOT NULL PRIMARY KEY,
    company_id character varying NOT NULL,
    title text NOT NULL DEFAULT 'Untitled Note',
    content text,
    category text DEFAULT 'general',
    is_pinned boolean DEFAULT false NOT NULL,
    last_edited_by character varying,
    last_edited_by_name text,
    created_by character varying NOT NULL,
    created_by_name text,
    created_at text NOT NULL,
    updated_at text
);

CREATE TABLE IF NOT EXISTS public.message_board_posts (
    id character varying DEFAULT gen_random_uuid() NOT NULL PRIMARY KEY,
    company_id character varying NOT NULL,
    title text NOT NULL,
    body text NOT NULL,
    category text DEFAULT 'general',
    is_pinned boolean DEFAULT false NOT NULL,
    is_announcement boolean DEFAULT false NOT NULL,
    reply_count integer DEFAULT 0 NOT NULL,
    posted_by character varying NOT NULL,
    posted_by_name text NOT NULL,
    created_at text NOT NULL,
    updated_at text
);

CREATE TABLE IF NOT EXISTS public.message_board_replies (
    id character varying DEFAULT gen_random_uuid() NOT NULL PRIMARY KEY,
    post_id character varying NOT NULL,
    company_id character varying NOT NULL,
    body text NOT NULL,
    posted_by character varying NOT NULL,
    posted_by_name text NOT NULL,
    created_at text NOT NULL,
    updated_at text
);

CREATE TABLE IF NOT EXISTS public.checkin_questions (
    id character varying DEFAULT gen_random_uuid() NOT NULL PRIMARY KEY,
    company_id character varying,
    question text NOT NULL,
    frequency text NOT NULL,
    scheduled_days text[],
    scheduled_time text DEFAULT '09:00',
    recipient_ids text[],
    recipient_type text DEFAULT 'all_agency',
    is_active boolean DEFAULT true NOT NULL,
    created_by character varying NOT NULL,
    next_send_at text,
    last_sent_at text,
    created_at text NOT NULL
);

CREATE TABLE IF NOT EXISTS public.checkin_responses (
    id character varying DEFAULT gen_random_uuid() NOT NULL PRIMARY KEY,
    question_id character varying NOT NULL,
    company_id character varying,
    responder_id character varying NOT NULL,
    responder_name text NOT NULL,
    response text NOT NULL,
    sent_at text NOT NULL,
    responded_at text NOT NULL,
    created_at text NOT NULL
);

CREATE TABLE IF NOT EXISTS public.hill_charts (
    id character varying DEFAULT gen_random_uuid() NOT NULL PRIMARY KEY,
    company_id character varying NOT NULL,
    title text NOT NULL,
    description text,
    items text NOT NULL DEFAULT '[]',
    created_by character varying NOT NULL,
    created_at text NOT NULL,
    updated_at text
);

=== FEATURE 4: HILL CHARTS ===

Add "Hill Chart" as a sub-tab under the Work section of company detail page.

THE VISUAL (rendered as SVG):
  Width: full container, Height: 280px
  Draw a smooth bell curve from left to right
  Left half labeled: "Figuring It Out" (gray italic)
  Right half labeled: "Making It Happen" (gray italic)
  X-axis labels: "Just Started" (0) | "Over the Hill" (50) | "Done" (100)
  
  Each work item = colored circle (24px) positioned on the curve
    X position = item.position mapped to curve x coordinate
    Y position = the Y value of the bell curve at that X
    Circle shows initials of item label (first 2 chars)
    Hover tooltip: full label + note + "last moved X ago"

BELOW THE CHART — item list:
  For each item:
    [Color swatch] [Label text] [Phase badge: "Uphill" or "Downhill"] 
    [Slider 0-100] [Note text input] [Delete ×]
  
  Phase is automatic: position < 50 = "Uphill 🔺", position >= 50 = "Downhill 🔻"
  Dragging slider → moves dot on chart in real time
  On slider release → PATCH /api/hill-charts/:id with updated items JSON

CREATING A HILL CHART:
  "New Hill Chart" button → modal:
    Title (e.g. "June Campaign", "Q3 SEO Project")
    Add work items (tag-style input — type name, press Enter to add)
    Each item gets a random color from a preset palette
    "Create" → saves to hill_charts table, all items start at position 5

PLACEMENT:
  Company Work tab → Hill Charts sub-tab
  Admin dashboard → mini hill chart widget per company 
    (shows latest hill_chart for each company, 
    read-only thumbnail with dot positions only)
